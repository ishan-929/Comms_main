#include <Arduino.h>
#include <driver/i2s.h>
#include <esp_now.h>
#include <WiFi.h>

#define I2S_PORT I2S_NUM_0
const int sck = 18;
const int ws = 19;
const int sd = 23;

typedef struct struct_message {
    int32_t audioBuffer[60];
} struct_message;
struct_message data;

void onDataRecieve(const uint8_t *mac, const uint8_t *incomingBytes, int len) {
    if (len == sizeof(data)) {
        memcpy(&data, incomingBytes, len);
        Serial.print("First 8 values: ");
        for (int i = 0; i < 8; i++) {
            Serial.print(data.audioBuffer[i]);
            Serial.print(' ');
        }
        Serial.println();

        size_t bytesWritten = 0;
        i2s_write(I2S_PORT, data.audioBuffer, sizeof(data.audioBuffer), &bytesWritten, 10 / portTICK_PERIOD_MS);
    }
}

void setup() {
    Serial.begin(115200);
    delay(1000);
    WiFi.mode(WIFI_STA);
    esp_now_init();
    esp_now_register_recv_cb(onDataRecieve);
    i2s_config_t config = {
        .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_TX),
        .sample_rate = 16000,
        .bits_per_sample = I2S_BITS_PER_SAMPLE_32BIT,
        .channel_format = I2S_CHANNEL_FMT_ONLY_RIGHT,
        .communication_format = (I2S_COMM_FORMAT_STAND_I2S),
        .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
        .dma_buf_count = 8,
        .dma_buf_len = 64,
    };
    i2s_driver_install(I2S_PORT,&config,0,NULL);
    i2s_pin_config_t pins = {
        .bck_io_num = sck,
        .ws_io_num = ws,
        .data_out_num = sd,
        .data_in_num = -1,
    };
    i2s_set_pin(I2S_PORT, &pins);
}
void loop() {

}
